## Install

::code-group

```bash [pnpm]
pnpm add comark-docs
```

```bash [npm]
npm install comark-docs
```

::

## Extend

```ts [nuxt.config.ts]
export default defineNuxtConfig({
  extends: ['comark-docs'],
  site: {
    url: 'https://docs.example.com',
    name: 'My Project',
  },
})
```
